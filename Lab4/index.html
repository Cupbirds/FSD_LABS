Task 2

Code 
function isPrime(num) {
  if (num < 2) return false;
  for (let i = 2; i * i <= num; i++) {
    if (num % i === 0) return false;
  }
  return true;
}

let givenPrime = 15;
let next = givenPrime + 1;
while (!isPrime(next)) {
  next++;
}
Output
Given number: 15
Next prime: 17

function createPhoneNumber(numbers) {
  let area = "" + numbers[0] + numbers[1] + numbers[2];
  let prefix = "" + numbers[3] + numbers[4] + numbers[5];
  let line = "" + numbers[6] + numbers[7] + numbers[8] + numbers[9];

}

Task 4

Code 
function roundMe(...args) {
  if (args.length === 0) return 0;
  if (args.length === 1) return Math.round(args[0]);
  let result = [];
  for (let i = 0; i < args.length; i++) {
    result.push(Math.round(args[i]));
  }
  return result;
}
Output
roundMe() → 0
roundMe(4.7) → 5
roundMe(4.7, 4.4) → [5, 4]

function abs(...args) {
  if (args.length === 0) return 0;

ceil(4.1) → 5

Task 6

Code 
function sumOfMultiples(x, y, z) {
  let sum = 0;
  for (let i = 1; i < z; i++) {
    if (i % x === 0 || i % y === 0) {
      sum += i;
    }
  }
  return sum;
}
Output
sumOfMultiples(3, 5, 10) → 23
